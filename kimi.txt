let i = 0, score = 0, locked = false;

function show(id){ ["scr-start","scr-game","scr-end"].forEach(s => document.getElementById(s).style.display = s===id ? "block" : "none"); }
function start(){ i = 0; score = 0; show("scr-game"); render(); }
function render(){
  locked = false;
  document.getElementById("qnum").textContent = "Вопрос " + (i+1) + " из " + Q.length;
  document.getElementById("score").textContent = score;
  document.getElementById("prog").style.width = (i / Q.length * 100) + "%";
  document.getElementById("qtext").textContent = Q[i].t;
  document.getElementById("fb").style.display = "none";
  ["bD","bI"].forEach(b => { const el = document.getElementById(b); el.disabled = false; el.classList.remove("ok","bad"); });
}
function answer(pick){
  if (locked) return; locked = true;
  const q = Q[i], right = pick === q.d;
  if (right) score++;
  const pickedBtn = pick === 1 ? "bD" : "bI", otherBtn = pick === 1 ? "bI" : "bD";
  document.getElementById(pickedBtn).classList.add(right ? "ok" : "bad");
  document.getElementById(otherBtn).disabled = true;
  const v = document.getElementById("fbVerdict");
  v.textContent = right ? "Верно!" : "Не совсем.";
  v.style.color = right ? "#2e9e5b" : "#d64545";
  const chip = q.d === 1 ? '<span class="chip d">прямое воздействие</span>' : '<span class="chip i">косвенное воздействие</span>';
  document.getElementById("fbText").innerHTML = "Это среда " + chip + " — " + q.e;
  document.getElementById("score").textContent = score;
  document.getElementById("fb").style.display = "block";
}
function next(){
  i++;
  if (i < Q.length) render(); else {
    show("scr-end");
    document.getElementById("finalScore").textContent = score + " / " + Q.length;
    let verdict = "Неплохо — факторы запомнятся после пары.", hint = "Совет: перед ответом спроси себя — можно ли этим фактором управлять напрямую?";
    if (score >= 19) { verdict = "Отличный аналитик рисков!"; hint = "Можешь смело вести этот блок на паре сам."; }
    else if (score >= 15) { verdict = "Уверенное понимание среды."; hint = "Запомни формулу: прямое — можно управлять, косвенное — только подстраиваться."; }
    else if (score >= 10) { verdict = "База есть, но стоит повторить схемы."; hint = "Косвенные факторы — экономика, политика, НТП, социо-культура, международная обстановка."; }
    document.getElementById("finalVerdict").textContent = verdict;
    document.getElementById("finalHint").textContent = hint;
  }
}
